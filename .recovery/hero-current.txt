  function inferLayerRole(layer){

  function isHeroCardDesign(design=activeDesign()){
    return String(design?.templateId||'')===HERO_CARD_TEMPLATE_ID;
  }

  function isHeroCardEditableTextLayer(layer){
    if(!layer||layer.type!=='text')return false;
    return HERO_CARD_TEXT_KEYS.includes(String(layer.textKey||''));
  }

  function isHeroCardEditablePhotoLayer(layer){
    if(!layer||layer.type!=='photo')return false;
    return String(layer.role||'')==='hero-portrait' || String(layer.name||'').toLowerCase().includes('hero portrait');
  }
